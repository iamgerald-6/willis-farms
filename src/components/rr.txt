@bp.get("/data/sql")
def data_sql():
  sql = """
      ;WITH maxd AS (
    SELECT MAX(TRY_CONVERT(date, [As of Date])) AS max_date
    FROM [DBG_DATA].[dbo].[t_KRI_DATA]
)
SELECT
    t.[KRI ID]              AS relatedEntityId,
    t.[Key Risk Indidcator]            AS metricName,
    t.[CurrentMth_Numeric] AS metricValue,
    t.[As of Date]          AS observedAt,
    t.[KRI Standard]        AS kriStandard,
    t.[Risk Type]           AS riskType,
    t.[Warning_Numeric]      AS warningLimit,
    t.[Warning_Symbol] AS warningLimitOperator,
    t.[Escalation_Numeric] AS escalationLimit,
    t.[Escalation_Symbol] AS escalationLimitOperator,
    t.[Threshold Limits]     AS thresholdLimit,
    t.[Breached KRIs]          AS statusBand,
    CASE
        WHEN t.[Breached KRIs] = '2' THEN 2
        WHEN t.[Breached KRIs] = '1'  THEN 1
        ELSE 0
    END AS breachLevel
FROM [DBG_DATA].[dbo].[t_KRI_DATA] t
WHERE
    t.[TOP_KRIs] = 1
    AND t.[Breached KRIs] > 0
    AND TRY_CONVERT(date, t.[As of Date]) 
        BETWEEN DATEADD(MONTH, -2, (SELECT max_date FROM maxd)) 
            AND (SELECT max_date FROM maxd)
ORDER BY
    breachLevel DESC,
    TRY_CONVERT(date, t.[As of Date]) DESC;

      """
  
  rows = run_query(sql)
  for r in rows:
    print(f"{r['relatedEntityId']} — {r['metricName']}: {r['metricValue']} | "
          f"status={r['statusBand']} | breachLevel={r['breachLevel']} | "
          f"asOf={r['observedAt']}")
  return jsonify({
        "meta": {"rows": len(rows), "window": "last_12_months_from_max_AsOfDate"},
        "data": {"source": "KRI", "rows": rows}
    })


@bp.post("/recommendations")
def post_recommendations():
    payload = request.get_json(force=True)
    items = payload if isinstance(payload, list) else [payload]
    validated = []
    for it in items:
        try:
            r = Recommendation(**it)
            rec = {
                "source": r.source,
                "relatedEntityId": r.relatedEntityId,
                "metricName": r.metricName,
                "metricValue": r.metricValue,
                "recommendationText": r.recommendationText,
                "actionType": r.actionType,
                "confidence": r.confidence,
                "riskType": r.riskType,
                "referenceTimestamp": r.referenceTimestamp,
                "observedAt": getattr(r, "observedAt", None) or datetime.today().date().isoformat(),
                "metadata": r.metadata or {},
                "postMitigationValue": getattr(r, "postMitigationValue", None)
            }
            validated.append(normalize_record(rec))

        except ValidationError as ve:
            return jsonify({"error": "validation_failed", "detail": ve.errors()}), 400
    inserted = insert_recommendations(validated)
    return jsonify({"inserted": inserted}), 201


@bp.post("/gpt/run")
def gpt_run():
    sql = """
WITH maxd AS (
    SELECT MAX(CONVERT(date, [As of Date])) AS max_date
    FROM v_KRI_Merge_values
    WHERE TOP_KRIs = 1
)
SELECT
    v.[KRI ID]          AS relatedEntityId,
    v.KRI_Name          AS metricName,

    -- 🔹 Add PFI name only for SOL (KRI-05112)
    CASE
        WHEN v.[KRI ID] = 'KRI-05112' THEN v.PFI
        ELSE NULL
    END AS pfiName,

    -- 🔹 Main numeric value the LLM should reason on
    CASE
        WHEN v.[KRI ID] = 'KRI-05112' THEN v.Sol_Limit                                      -- SOL
        WHEN v.[KRI ID] = 'KRI-05203' THEN v.Cap_ratio                                      -- CAR
        WHEN v.[KRI ID] = 'KRI-05204' THEN v.Leverage_Ratio                                 -- Leverage
        WHEN v.[KRI ID] = 'KRI-05207' AND v.KRI_Name LIKE '%EUR%' THEN v.NOP_EUR_Perc       -- NOP EUR
        WHEN v.[KRI ID] = 'KRI-05207' AND v.KRI_Name LIKE '%USD%' THEN v.NOP_USD_Perc       -- NOP USD
        ELSE v.CurrentMth_Val                                                               -- Other KRIs
    END AS metricValue,

    v.[As of Date]      AS observedAt,
    v.[Risk Type]       AS riskType,

    -- Thresholds (raw values, as required)
    v.Warning_Numeric       AS warningLimit,
    v.[Warning_Symbol]      AS warningLimitOperator,
    v.Escalation_Numeric    AS escalationLimit,
    v.[Escalation_Symbol]   AS escalationLimitOperator,

    -- Final merged status band
    v.KRI_Status        AS statusBand,

    CASE
        WHEN v.KRI_Status = 'Breached' THEN 2
        WHEN v.KRI_Status = 'Warning'  THEN 1
        ELSE 0
    END AS breachLevel

FROM v_KRI_Merge_values AS v
CROSS JOIN maxd
WHERE
    v.TOP_KRIs = 1
    AND v.KRI_Status IN ('Breached', 'Warning')   -- ignore Safe
    AND YEAR(v.[As of Date]) = YEAR(maxd.max_date)
    AND MONTH(v.[As of Date]) = 9                 -- you filtered month 1 (January)
ORDER BY
    breachLevel DESC,
    CONVERT(date, v.[As of Date]) DESC;
       """

    rows = run_query(sql)
    compact = {"source": "KRI", "window": "year_2025", "rows": rows}
    recs = call_gpt(compact)

    validated, errors = [], []

    for it in recs:
      try:
        r = Recommendation(**it)
        rec = {
            "source": r.source,
            "relatedEntityId": r.relatedEntityId,
            "metricName": r.metricName,
            "metricValue": r.metricValue,
            "recommendationText": r.recommendationText,
            "rootCause": r.rootCause,
            "actionType": r.actionType,
            "confidence": r.confidence,
            "riskType": r.riskType,
            "referenceTimestamp": r.referenceTimestamp,
            "observedAt": getattr(r, "observedAt", None)
                or next((row["observedAt"] for row in rows if row["relatedEntityId"] == r.relatedEntityId), None),
            "metadata": r.metadata or {},
            "postMitigationValue": getattr(r, "postMitigationValue", None)
        }
        validated.append(normalize_record(rec))
      except Exception as e:
        errors.append({"item": it, "error": str(e)})
    count = insert_recommendations(validated) if validated else 0
    return jsonify({"generated": len(recs), "inserted":count,  "errors": errors,  "recommendations": validated })


@bp.post("/gpt/liquidity")
def gpt_liquidity():
    # For the first run: January → max date; later you can switch to max-date-only
    sql = """
       SELECT TOP 1
    CAST(MonthEndDate AS date) AS observedAt,
    Total_Liquid_Assets AS totalLiquidAssets,
    Current_Liabilities AS currentLiabilities,
    Liquid_Ratio AS liquidRatio
FROM [DBG_DATA].[dbo].[t_CalculateCapitalAdequacyRatios]
ORDER BY CAST(MonthEndDate AS date) DESC;

    """

    rows = run_query(sql)

    # Transform into the long format the prompt expects: metricName + metricValue
    series = []
    for r in rows:
        obs = r["observedAt"]
        series.append({
            "observedAt": obs,
            "metricName": "Liquid_Ratio",
            "metricValue": r["liquidRatio"],
        })
        series.append({
            "observedAt": obs,
            "metricName": "Current_Liabilities",
            "metricValue": r["currentLiabilities"],
        })

    compact = {
        "messages": [
            {"role": "system", "content": LIQUIDITY_PROMPT},
            {"role": "user", "content": json.dumps(series, ensure_ascii=False)},
        ]
    }

    result = call_gpt(compact)

    items = result if isinstance(result, list) else [result]

    formatted = []
    for it in items:
        formatted.append({
            "observedAt": it.get("observedAt"),
            "metricName": it.get("metricName"),
            "metricValue": it.get("metricValue"),
            "rootCauses": it.get("rootCauses", []),
            "recommendations": it.get("recommendations", []),
        })

    inserted = insert_liquidity_recommendations(formatted)

    return jsonify({
        "generated": len(formatted),
        "inserted": inserted,
        "items": formatted,
    })


@bp.post("/gpt/summary")
def gpt_summary():
    sql = """
            ;WITH maxd AS (
            SELECT MAX(TRY_CONVERT(date, [As of Date])) AS max_date
            FROM [DBG_DATA].[dbo].[t_KRI_DATA]
        )
        SELECT
            t.[KRI ID]                AS kriId,
            t.[Key Risk Indidcator]              AS kriName,
            t.[Adjusted Current Mth]  AS adjustedCurrentMth,
            t.[Risk Type]             AS riskType,
            t.[Warning_Numeric]      AS warningLimit,
            t.[Warning_Symbol] AS warningLimitOperator,
            t.[Escalation_Numeric] AS escalationLimit,
            t.[Escalation_Symbol] AS escalationLimitOperator,
            t.[Threshold Limits]     AS thresholdLimit,
            t.[Breached KRIs]          AS statusBand,
            TRY_CONVERT(date, t.[As of Date]) AS asOfDate,
            CASE WHEN t.[Breached KRIs] = 'Breached' THEN 1 ELSE 0 END AS isBreached,
            CASE WHEN t.[Breached KRIs] = 'Warning'  THEN 1 ELSE 0 END AS isWarning,
            CASE WHEN t.[ImpactBin_Col] = 3 AND t.[LikelihoodBin_Col] = 5 THEN 1 ELSE 0 END AS highImpactHighLikelihood
        FROM [DBG_DATA].[dbo].[t_KRI_DATA] t
        CROSS JOIN maxd
        WHERE TRY_CONVERT(date, t.[As of Date]) = maxd.max_date
        AND t.[TOP_KRIs] = 1
        ORDER BY t.[Risk Type], t.[Key Risk Indidcator];
    """
    
    rows = run_query(sql)
    compact = {"source": "KRI", "window": "current_month", "rows": rows}
    summary_payload = {
        "messages": [
            {"role": "system", "content": summary_prompt},
            {"role": "user", "content": json.dumps(compact, ensure_ascii=False)}
        ]
    }

    summary_text = call_gpt(summary_payload)

                <br><br>
                <p style="font-size:14px; color:#333; line-height:1.6;">
                    For more information, click the link to view the dashboard:
                    <a href="{dashboard_link}" style="color:#0078D4; text-decoration:underline;">
                        {dashboard_name}
                    </a>
                </p>
            """
    else:
        email_body += """
            <br><br>
            <p style="font-size:14px; color:#999; line-height:1.6;">
                Dashboard link currently unavailable.
        </p>
    """

    email_body += """
        <br><br>
        <p style="font-size:14px; color:#333; line-height:1.6;">
            Best regards,<br>
            <strong>InsightView Monitoring System</strong>
        </p>
        <p style="font-size:12px; color:#777; line-height:1.4;">
            This is a system-generated message from InsightView.<br>
            Please do not reply to this email.
        </p>
    """

    recipients = get_llm_suggestive_recipients()

    subject = f"DBG KRI Summary Report – {as_of_date}"

    success = send_summary_email(subject, email_body, recipients, is_html=True)

    if success:
        # Mark as emailed

            UPDATE [DBG_DATA].[dbo].[t_insightView_Email_Summaries]

        print(f"⚠️ Email failed for summary {as_of_date}. Will remain IsEmailed = 0.")

    return jsonify({
        "summary_saved": True,
        "emailed": success,
        "asOfDate": as_of_date,
        "summary": summary_text
    })

def run_query(sql: str, params: dict | None = None) -> list[dict]:
    with ENGINE.begin() as conn:
        res = conn.execute(text(sql), params or {})
        cols = res.keys()
        rows = []
        for row in res.fetchall():
            row_dict = {}
            for col, val in zip(cols, row):
                if isinstance(val, (date, datetime)):
                    row_dict[col] = val.isoformat()
                elif isinstance(val, Decimal):
                    row_dict[col] = float(val)
                else:
                    row_dict[col] = val
            rows.append(row_dict)
        return rows


def normalize_record(r: dict) -> dict:
    rec = dict(r)
    # ReferenceTimestamp → datetime
    if rec.get("referenceTimestamp"):
        try:
            rec["referenceTimestamp"] = datetime.fromisoformat(
                rec["referenceTimestamp"].replace("Z", "")
            )
        except Exception:
            rec["referenceTimestamp"] = None
    # ObservedAt → date
    if rec.get("observedAt"):
        try:
            rec["observedAt"] = datetime.fromisoformat(
                str(rec["observedAt"]).replace("Z", "")
            ).date()
        except Exception:
            try:
                rec["observedAt"] = date.fromisoformat(str(rec["observedAt"]))
            except Exception:
                print(f"[WARN] Could not parse observedAt: {rec['observedAt']}, defaulting to today")
                rec["observedAt"] = date.today()

    # Metadata → JSON string
    if isinstance(rec.get("metadata"), dict):
        rec["metadata"] = json.dumps(rec["metadata"], ensure_ascii=False)
    return rec


def is_latest_kri_processed() -> bool:
    """
    Returns True if the latest KRI 'As of Date' already exists in Recommendations.
    """
    sql = """
    ;WITH kri_max AS (
        SELECT MAX(TRY_CONVERT(date, [As of Date])) AS kri_latest
        FROM [DBG_DATA].[dbo].[t_KRI_DATA]
    ),
    rec_check AS (
        SELECT MAX(TRY_CONVERT(date, ObservedAt)) AS rec_latest
        FROM [DBG_DATA].[dbo].[t_insightView_Recommendations]
    )
    SELECT
        CASE
            WHEN kri_latest <= ISNULL(rec_latest, '1900-01-01') THEN 1
            ELSE 0
        END AS is_processed,
        kri_latest, rec_latest
    FROM kri_max CROSS JOIN rec_check;
    """
    result = run_query(sql)
    if result:
        row = result[0]
        print(f"Latest KRI date: {row['kri_latest']}, Latest Recommendation: {row['rec_latest']}")
        return row["is_processed"] == 1
    return False


def is_latest_summary_processed() -> bool:
    """
    Checks if the latest KRI 'As of Date' is already processed (emailed) in t_insightView_Email_Summaries.
    Returns True if already emailed (skip), False if new or pending (trigger send).
    """
    sql = """
    ;WITH kri_max AS (
        SELECT MAX(TRY_CONVERT(date, [As of Date])) AS kri_latest
        FROM [DBG_DATA].[dbo].[t_KRI_DATA]
    ),
    email_check AS (
        SELECT MAX(TRY_CONVERT(date, AsOfDate)) AS email_latest,
               MAX(CASE WHEN IsEmailed = 1 THEN 1 ELSE 0 END) AS emailed_flag
        FROM [DBG_DATA].[dbo].[t_insightView_Email_Summaries]
    )
    SELECT 
        CASE 
            WHEN kri_latest <= ISNULL(email_latest, '1900-01-01') AND emailed_flag = 1 THEN 1
            ELSE 0
        END AS is_processed,
        kri_latest, email_latest, emailed_flag
    FROM kri_max CROSS JOIN email_check;
    """

    result = run_query(sql)
    if result:
        row = result[0]
        print(
            f"📅 Latest KRI Date: {row['kri_latest']}, Email Summary: {row['email_latest']} | Emailed Flag: {row['emailed_flag']}")
        return row["is_processed"] == 1
    return False


def is_latest_liquidity_processed() -> bool:
    """
    Returns True ONLY if the latest liquidity MonthEndDate already exists 
    as ObservedAt in t_recommendations_liquidity.
    """
    sql = """
    ;WITH liq_max AS (
        SELECT MAX(CAST(MonthEndDate AS date)) AS liq_latest
        FROM [DBG_DATA].[dbo].[t_CalculateCapitalAdequacyRatios]
    ),
    rec_max AS (
        SELECT MAX(CAST(ObservedAt AS date)) AS rec_latest
        FROM [DBG_DATA].[dbo].[t_recommendations_liquidity]
    )
    SELECT 
        CASE 
            WHEN rec_latest IS NOT NULL 
                 AND liq_latest = rec_latest THEN 1 
            ELSE 0 
        END AS is_processed,
        liq_latest,
        rec_latest
    FROM liq_max
    CROSS JOIN rec_max;
    """

    result = run_query(sql)
    if result:
        row = result[0]
        print(f"Latest Liquidity Date: {row['liq_latest']}, "
              f"Latest Recommendation Date: {row['rec_latest']}")
        return row["is_processed"] == 1

    return False


def send_summary_email(subject: str, body: str, recipients: list[str], is_html: bool = True) -> bool:
    """
    Sends an email using Office 365 SMTP settings.
    Returns True if successful, False otherwise.
    """
    try:
        SMTP_SERVER = os.getenv("SMTP_SERVER", "smtp.office365.com")
        SMTP_PORT = int(os.getenv("SMTP_PORT", "587"))
        SMTP_EMAIL = os.getenv("SMTP_EMAIL")
        SMTP_PASSWORD = os.getenv("SMTP_PASSWORD")

        msg = MIMEMultipart("alternative")
        msg["From"] = SMTP_EMAIL
        msg["To"] = ", ".join(recipients)
        msg["Subject"] = subject

        plain_body = body.replace("<br>", "\n").replace("<div>", "").replace("</div>", "")
        msg.attach(MIMEText(plain_body, "plain"))
        if is_html:
            # wrap in minimal HTML for safe rendering
            html_body = f"<html><body>{body}</body></html>"
            msg.attach(MIMEText(html_body, "html"))

        with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
            server.starttls()
            server.login(SMTP_EMAIL, SMTP_PASSWORD)
            server.send_message(msg)

        print(f"📧 Email sent to: {', '.join(recipients)}")
        return True

    except Exception as e:
        print(f"[EMAIL ERROR] {e}")
        return False


def get_published_address(summary_type: str) -> tuple[str, str] | None:
    """
    Returns (DashboardName, publishedAddress) for the given summary type.
    Uses the t_PublishedDashboards table.
    """
    mapping = {
        "KRI": "Key Risk Indicator Overview",
        "Finance": "Financial Overview",
        "ESG": "ESG Dashboard",
        "Treasury": "Treasury Performance Dashboard"
    }

    dashboard_name = mapping.get(summary_type.upper())
    if not dashboard_name:
        return None

    sql = """
        SELECT TOP 1 DashboardName, publishedAddress
        FROM [DBG_DATA].[dbo].[t_PublishedDashboards]
        WHERE DashboardName = :dashboard_name;
    """
    rows = run_query(sql, {"dashboard_name": dashboard_name})
    if rows:
        return rows[0]["DashboardName"], rows[0]["publishedAddress"]
    return None


def auto_scheduler():
    """
    Scheduler that runs every Monday at 12:00 PM.
    If the latest KRI date already has recommendations → skip rec generation.
    If the latest KRI date already has an emailed summary → skip summary email.
    Otherwise, → trigger GPT generation for both.
    """
    last_run_date = None

    while True:
        now = datetime.now()

        if now.weekday() == 0 and now.hour >= 3:  # Monday = 0 (12 PM)
            if last_run_date != now.date():
                print("⏰ Monday 12:00 → Checking if new KRI data exists")

                # --- STEP 1: Check Recommendations ---
                if is_latest_kri_processed():
                    print("✅ Latest KRI data already has recommendations → Skipping GPT run.")
                else:
                    print("🧠 New KRI data detected — triggering GPT recommendation generation...")
                    try:
                        requests.post("http://127.0.0.1:" + PORT_NUMBER + "/gpt/run", timeout=700)
                        print("✅ Recommendations generated successfully.")
                    except Exception as e:
                        print(f"[RECOMMENDATION ERROR] {e}")

                # --- STEP 2: Check Email Summaries ---
                # try:
                #     if is_latest_summary_processed():
                #         print("✅ Latest summary already emailed → Skipping summary send.")
                #     else:
                #         print("🧾 New summary pending — triggering GPT summary and email send...")
                #         requests.post("http://127.0.0.1:8080/gpt/summary", timeout=500)
                #         print("✅ Summary generated and emailed successfully.")
                # except Exception as e:
                #     print(f"[SUMMARY ERROR] {e}")

                try:
                    if not is_latest_liquidity_processed():
                        requests.post("http://127.0.0.1:" + PORT_NUMBER + "/gpt/liquidity", timeout=700)
                    else:
                        print("✅ Latest liquidity data already has recommendations → Skipping GPT run.")
                except Exception as e:
                    print(f"[Liquidity ERROR] {e}")

                # --- STEP 3: Sleep after completion ---
                last_run_date = now.date()
                print("💤 Sleeping for 7 days (until next Monday)...")
                time.sleep(7 * 24 * 3600)

            else:
                time.sleep(3600)

        elif now.weekday() == 2 and now.hour < 17:
            print("🕛 Monday but before 12:00 → Sleeping 1 hour")
            time.sleep(3600)
        else:
            print("📆 Not Monday → Sleeping 1 day")
            time.sleep(86400)


def insert_recommendations(recs: list[dict]) -> int:
    """
    Insert a batch of recommendations into dbo.Recommendations.
    Expects each dict in `recs` to have keys that match the placeholders.
    """
    if not recs:
        return 0
    clean_recs = [normalize_record(r) for r in recs]
    sql = text("""
    INSERT INTO dbo.t_insightView_Recommendations (
        Source,
        RelatedEntityId,
        MetricName,
        MetricValue,
        RecommendationText,
        RootCause,
        ActionType,
        RiskType,
        Confidence,
        ReferenceTimestamp,
        ObservedAt,
        Metadata,
        PostMitigationValue
    )
    VALUES (
        :source,
        :relatedEntityId,
        :metricName,
        :metricValue,
        :recommendationText,
        :rootCause,
        :actionType,
        :riskType,               
        :confidence,
        :referenceTimestamp,
        :observedAt,
        :metadata,
        :postMitigationValue
    )
""")

    with ENGINE.begin() as conn:
        conn.execute(sql, clean_recs)

    return len(clean_recs)


def insert_summary(summary_text: str, summary_type: str, as_of_date: str | date) -> int:
    """
    Inserts a GPT-generated summary into dbo.t_insightView_Summaries.
    Automatically marks it as not emailed (IsEmailed = 0).
    Returns the number of inserted rows (should be 1 on success).
    """
    sql = text("""
        INSERT INTO dbo.t_insightView_Email_Summaries (SummaryType, SummaryText, AsOfDate, IsEmailed)
        VALUES (:summaryType, :summaryText, :asOfDate, 0);
    """)

    with ENGINE.begin() as conn:
        conn.execute(sql, {
            "summaryType": summary_type,
            "summaryText": summary_text,
            "asOfDate": as_of_date
        })

    print(f"Summary saved → Type: {summary_type}, Date: {as_of_date}")
    return 1


def insert_liquidity_recommendations(recs: list[dict]) -> int:
    """
    Insert a batch of liquidity recommendations into dbo.t_recommendations_liquidity.
    rootCauses and recommendations are stored as JSON strings.
    """
    if not recs:
        return 0

    payload = []
    for r in recs:
        # --- FIX: Strip timestamp -> keep only YYYY-MM-DD ---
        obs = r.get("observedAt")
        if isinstance(obs, str) and "T" in obs:
            obs = obs.split("T")[0]

        payload.append({
            "observedAt": obs,
            "metricName": r.get("metricName"),
            "metricValue": r.get("metricValue"),
            "rootCauses": json.dumps(r.get("rootCauses", []), ensure_ascii=False),
            "recommendations": json.dumps(r.get("recommendations", []), ensure_ascii=False),
        })

    sql = text("""
        INSERT INTO dbo.t_recommendations_liquidity (
            ObservedAt,
            MetricName,
            MetricValue,
            RootCauses,
            Recommendations
        )
        VALUES (
            :observedAt,
            :metricName,
            :metricValue,
            :rootCauses,
            :recommendations
        )
    """)

    with ENGINE.begin() as conn:
        conn.execute(sql, payload)

    return len(payload)

def call_gpt(compact_payload: dict) -> any:
    """
    Handles both recommendation JSON requests and text summaries.
    If the compact_payload has `messages`, treat it as a direct GPT request (used for summaries).
    Otherwise, use SYSTEM_PROMPT + KRI JSON logic.
    """
    url = f"{OPENAI_BASE}/chat/completions"
    headers = {"Authorization": f"Bearer {OPENAI_API_KEY}", "Content-Type": "application/json"}

    if "messages" in compact_payload:
        body = {"model": OPENAI_MODEL, **compact_payload, "temperature": 0.1}
    else:
        body = {
            "model": OPENAI_MODEL,
            "messages": [
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": json.dumps(compact_payload, ensure_ascii=False)}
            ],
            "temperature": 0.1
        }
    r = requests.post(url, headers=headers, json=body, timeout=300)
    r.raise_for_status()
    content = r.json()["choices"][0]["message"]["content"].strip()
    if content.startswith("[") or content.startswith("{"):
        try:
            parsed = json.loads(content)
            return parsed if isinstance(parsed, list) else parsed.get("recommendations", [])
        except Exception:
            return []
    else:
        return content
